function portraits(c){const l=layout(c),pair=DATA.posterSets[c.posterSet];$('portraits').replaceChildren();$('portrait-labels').replaceChildren();for(const side of [0,1]){const pw=l.left,left=side?l.right:0,key=side?'alliance':'empire',asset=pair[side],scale=Math.max(pw/asset.width,720/asset.height),w=asset.width*scale,x=left-(w-pw)*(side?.5:.56),natural=side?left-x:x+w-l.left,extension=Math.max(0,Math.min(natural-1,natural*c.fade/100));const clip=$((side?'right':'left')+'-portrait-clip').firstElementChild;clip.setAttribute('x',left);clip.setAttribute('width',pw);const group=node('g',{'clip-path':`url(#${side?'right':'left'}-portrait-clip)`,class:'portrait-panel','data-side':key,'data-set':c.posterSet,'data-natural-overhang':natural,'data-extension':extension});$('portraits').append(group);image(group,asset,x,(720-asset.height*scale)/2,w);
 const extClip=$((side?'right':'left')+'-extension-clip');extClip.replaceChildren();if(extension>0){for(const band of [{y:0,h:l.topH},{y:l.bottomY,h:l.bottomH}])extClip.append(node('rect',{x:side?left-extension:l.left-1,y:band.y,width:extension+1,height:band.h}));const ext=node('g',{'clip-path':`url(#${side?'right':'left'}-extension-clip)`,mask:`url(#portrait-mask-${side?'right':'left'})`,class:'portrait-extension','data-side':key});$('portraits').append(ext);image(ext,asset,x,(720-asset.height*scale)/2,w)}if(c.posterSet!=='v1')continue;
 const labels=node('g',{'clip-path':`url(#${side?'right':'left'}-portrait-clip)`});$('portrait-labels').append(labels);labels.append(node('rect',{x:left,y:c.quoteY-65,width:pw,height:130,fill:'url(#text-vignette)'}));const quote=node('text',{id:'quote-'+key,x:left+pw/2,'text-anchor':'middle',class:'portrait-quote','data-inside-picture':'true'});const lines=quoteLines(DATA.quotes['poster.quote.'+key][c.locale],c.locale);lines.forEach((line,i)=>{const span=node('tspan',{x:left+pw/2,y:c.quoteY-(lines.length-1)*9+i*18});span.textContent=line;quote.append(span)});labels.append(quote);const name=node('text',{id:'name-'+key,x:left+pw/2,y:c.quoteY+30,'text-anchor':'middle',class:'portrait-name','data-inside-picture':'true'});name.textContent=DATA.quotes['poster.name.'+key][c.locale];labels.append(name)}}
function junctions(c){const l=layout(c);for(const [side,key]of[['left','empire'],['right','alliance']]){const panel=document.querySelector('.portrait-panel[data-side="'+key+'"]'),extension=Number(panel.dataset.extension),gradient=$('portrait-feather-'+side);gradient.setAttribute('x1',side==='left'?l.left:l.right-extension);gradient.setAttribute('x2',side==='left'?l.left+extension:l.right);if(extension>0)panel.setAttribute('mask','url(#portrait-mask-'+side+')');else panel.removeAttribute('mask')}$('junctions').replaceChildren()}
