function layout(c){const s=c.gameScale/100,w=960*s,h=720*s,left=(1280-w)/2,gy=(720-h)/2;return {s,w,h,left,right:left+w,gy,topH:gy+62*s,bottomY:gy+627*s,bottomH:720-(gy+627*s)}}
function singleFlagship(c,side){const l=layout(c),group=$(side?'alliance-row':'empire-row'),asset=DATA.heroes[side],band=side?{y:0,h:l.topH}:{y:l.bottomY,h:l.bottomH},w=l.w*(side?c.topSize:c.bottomSize)/100,h=w*asset.hullHeight/asset.hullWidth,scale=w/asset.hullWidth,cy=band.y+band.h/2+(side?c.topY:c.bottomY),angle=side?-12:-10;
 group.replaceChildren();const g=node('g',{class:'flagship hero','data-key':side?'46BD':'45B1','data-facing':side?'left':'right','data-width':w,'data-hull':'large-closeup',transform:`translate(${l.left+l.w/2-(side?0:16)} ${cy}) rotate(${angle})`});group.append(g);
 image(g,asset,-w/2-asset.hullX*scale,-h*(side?.45:.65)-asset.hullY*scale,asset.width*scale);
}
function render(){const c=settings(),l=layout(c);for(const k of ['topSize','bottomSize','topY','bottomY','quoteY','gameScale','fade'])$(k+'Value').textContent=c[k]+(['topSize','bottomSize','gameScale'].includes(k)?'%':' px');$('scene').classList.toggle('show-guides',c.guidesToggle);for(const[k,v]of Object.entries({x:l.left,y:l.gy,width:l.w,height:l.h}))$('game-canvas').setAttribute(k,v);for(const[id,y,h]of[['top-clip',0,l.topH],['bottom-clip',l.bottomY,l.bottomH]]){const r=$(id).firstElementChild;for(const[k,v]of Object.entries({x:l.left,y,width:l.w,height:h}))r.setAttribute(k,v)}for(const[id,y,h]of[['top-bg',0,l.topH],['bottom-bg',l.bottomY,l.bottomH]])for(const[k,v]of Object.entries({x:l.left,y,width:l.w,height:h}))$(id).setAttribute(k,v);$('guides').replaceChildren();for(const band of [{y:0,h:l.topH},{y:l.bottomY,h:l.bottomH}])$('guides').append(node('rect',{x:l.left,y:band.y,width:l.w,height:band.h}));portraits(c);singleFlagship(c,1);singleFlagship(c,0);junctions(c);effects(time);window.proposalSettings=c;window.proposalLayout=l}
