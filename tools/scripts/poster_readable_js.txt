function layout(c){const s=c.gameScale/100,w=960*s,h=720*s,left=(1280-w)/2,gy=(720-h)/2;return {s,w,h,left,right:left+w,gy,topH:gy+62*s,bottomY:gy+627*s,bottomH:720-(gy+627*s)}}
function portraits(c){const l=layout(c);$('portraits').replaceChildren();$('portrait-labels').replaceChildren();for(const side of [0,1]){const pw=l.left,left=side?l.right:0,key=side?'alliance':'empire',asset=DATA.portraits[side],clip=$((side?'right':'left')+'-portrait-clip').firstElementChild;clip.setAttribute('x',left);clip.setAttribute('width',pw);const group=node('g',{'clip-path':`url(#${side?'right':'left'}-portrait-clip)`,class:'portrait-panel','data-side':key});$('portraits').append(group);const scale=Math.max(pw/asset.width,720/asset.height),w=asset.width*scale,x=left-(w-pw)*(side?.5:.56);image(group,asset,x,(720-asset.height*scale)/2,w);const labels=node('g',{'clip-path':`url(#${side?'right':'left'}-portrait-clip)`});$('portrait-labels').append(labels);labels.append(node('rect',{x:left,y:c.quoteY-65,width:pw,height:130,fill:'url(#text-vignette)'}));const quote=node('text',{id:'quote-'+key,x:left+pw/2,'text-anchor':'middle',class:'portrait-quote','data-inside-picture':'true'});const lines=quoteLines(DATA.quotes['poster.quote.'+key][c.locale],c.locale);lines.forEach((line,i)=>{const span=node('tspan',{x:left+pw/2,y:c.quoteY-(lines.length-1)*9+i*18});span.textContent=line;quote.append(span)});labels.append(quote);const name=node('text',{id:'name-'+key,x:left+pw/2,y:c.quoteY+30,'text-anchor':'middle',class:'portrait-name','data-inside-picture':'true'});name.textContent=DATA.quotes['poster.name.'+key][c.locale];labels.append(name)}}
function fleet(c,side){const l=layout(c),group=$(side?'alliance-row':'empire-row');group.replaceChildren();const ships=DATA.ships.filter(s=>s.side===side),band=side?{y:0,h:l.topH}:{y:l.bottomY,h:l.bottomH};const hero=DATA.heroes[side],heroH=(band.h-14)*c.size/100,heroW=heroH*hero.width/hero.height;for(let i=c.count-1;i>=1;i--){const near=1-i/(c.count-1),height=(side?50:75)*(1-c.depth/100*(1-near));const g=node('g',{'data-key':ships[i].key,'data-depth':near.toFixed(4),'data-facing':side?'left':'up',class:'flagship escort'});group.append(g);if(side){const available=l.w-heroW-200,offset=(i-1)/Math.max(1,c.count-2);const x=l.left+heroW+120+available*offset*c.spread/830,y=band.y+band.h*.48-height*.6+(1-near)*9+c.topY;image(g,ships[i],x,y,height*ships[i].width/ships[i].height,true)}else{const offset=(i-1)/Math.max(1,c.count-2),cx=l.left+70+(l.w-290)*offset*c.spread/830,cy=band.y+band.h*.53+(1-near)*13+c.bottomY;g.setAttribute('transform',`translate(${cx} ${cy}) rotate(-90)`);image(g,ships[i],-height/2,-height*ships[i].height/ships[i].width/2,height)}}
 const g=node('g',{'data-key':side?'46BD':'45B1','data-depth':'1','data-facing':side?'left':'up','data-camera':side?'low-angle':'upward',class:'flagship hero'});group.append(g);const x=side?l.left+55:l.right-105-heroW/2,y=band.y+(band.h-heroH)/2+(side?c.topY:c.bottomY);image(g,hero,x,y,heroW);g.dataset.hull='full';g.dataset.bandY=band.y;g.dataset.bandH=band.h;
}
function junctions(c){const l=layout(c),fade=c.fade;for(const [id,x1,x2]of[['seam-left',l.left-fade,l.left+12],['seam-right',l.right+fade,l.right-12]]){const g=$(id);g.setAttribute('x1',x1);g.setAttribute('x2',x2)}$('junctions').replaceChildren();if(fade>0)for(const band of [{y:0,h:l.topH},{y:l.bottomY,h:l.bottomH}]){for(const side of [0,1])$('junctions').append(node('rect',{x:side?l.right-12:l.left-fade,y:band.y,width:fade+12,height:band.h,fill:'url(#row-bg)',mask:`url(#seam-mask-${side?'right':'left'})`,'data-junction':side?'right':'left'}))}}
function render(){const c=settings(),l=layout(c);for(const k of ['count','size','depth','spread','topY','bottomY','quoteY','gameScale','fade'])$(k+'Value').textContent=c[k]+(k==='count'?' 艘':['depth','gameScale','size'].includes(k)?'%':' px');$('scene').classList.toggle('show-guides',c.guidesToggle);for(const [k,v]of Object.entries({x:l.left,y:l.gy,width:l.w,height:l.h}))$('game-canvas').setAttribute(k,v);for(const[id,y,h]of[['top-clip',0,l.topH],['bottom-clip',l.bottomY,l.bottomH]]){const r=$(id).firstElementChild;for(const[k,v]of Object.entries({x:l.left,y,width:l.w,height:h}))r.setAttribute(k,v)}for(const[id,y,h]of[['top-bg',0,l.topH],['bottom-bg',l.bottomY,l.bottomH]])for(const[k,v]of Object.entries({x:l.left,y,width:l.w,height:h}))$(id).setAttribute(k,v);$('guides').replaceChildren();for(const band of [{y:0,h:l.topH},{y:l.bottomY,h:l.bottomH}])$('guides').append(node('rect',{x:l.left,y:band.y,width:l.w,height:band.h}));portraits(c);fleet(c,1);fleet(c,0);junctions(c);effects(time);window.proposalSettings=c;window.proposalLayout=l}
